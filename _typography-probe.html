const s = (sel) => {
  const el = document.querySelector(sel);
  if (!el) return 'NOT FOUND';
  const cs = getComputedStyle(el);
  return cs.fontSize + ' / line-height: ' + cs.lineHeight;
};
console.table({
  'Body text':         s('body'),
  'Paragraph (p)':     s('p'),
  'H1 (hero)':         s('h1'),
  'H2 (section)':      s('h2'),
  'H3 (card)':         s('h3'),
  'Nav link':          s('.main-nav a, .nav-group > a'),
  'Footer link':       s('footer a'),
  'Small text (.eyebrow)': s('.eyebrow'),
  'Button':            s('button'),
});
